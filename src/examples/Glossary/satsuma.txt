// =====================================================
// GLOBAL SCALE + MATERIAL/FIT (shared)
// =====================================================
param coeff 3/4

param materialThickness0 3
param kerf0 0.20
param clearance0 0.15

param materialThickness (materialThickness0 * coeff)

// =====================================================
// GLOBAL OFFSETS (move each “module” away from others)
// =====================================================
// Body block center
param bodyOX 0
param bodyOY 0

// Head block (cube + connector) moved to the right
param headOX (260 * coeff)
param headOY 0

// Shell block moved to the left

// #####################################################
// 1) BODY + LEGS  (offset by bodyOX/bodyOY)

param body_servoW0 11.5 + kerf0 + clearance0

param s45  0.70710678

param s135  0.70710678

param s225 -0.70710678

  fill: false

  fill: false

  rotation: 135
  fill: false
}

  position: [bodyOX + legServoOffset*c225, bodyOY + legServoOffset*s225]

  fill: false
